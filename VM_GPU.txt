# ==========================================
sudo apt clean
sudo apt update
sudo apt install -y nvidia-driver-595-open
sudo reboot

# => Sau khi reboot, kết nối SSH lại vào VM

# ==========================================
# 2. CÀI ĐẶT CÔNG CỤ NỀN TẢNG (UV, GH, TMUX)
# ==========================================
# Cài đặt GitHub CLI và tmux (giúp giữ phiên train chống rớt mạng)
sudo apt update
sudo apt install -y gh tmux

# Cài đặt uv
curl -LsSf https://astral.sh/uv/install.sh | sh
source ~/.bashrc

# Đăng nhập GitHub
gh auth login
# -> Chọn: HTTPS -> Tokens -> Nhập token (tạo classic token với quyền repo, write)

git config --global user.email "....@gmail.com"

# ==========================================
# 3. CLONE CODE & TẠO MÔI TRƯỜNG ẢO
# ==========================================
# Clone trực tiếp nhánh DuyKhang:
git clone -b DuyKhang https://github.com/fedmed-FA26AI28/FedMed_20K.git
cd FedMed_20K

uv venv --python 3.10
source .venv/bin/activate

# Cài đặt PyTorch với CUDA 12.1 (đảm bảo nhận GPU):
uv pip install torch torchvision --index-url https://download.pytorch.org/whl/cu121

# Cài đặt các thư viện dự án:
uv pip install -r requirements.txt
# (hoặc: pip install -r requirements.txt)

# Kiểm tra nhận diện GPU:
python -c "import torch; print('CUDA Available:', torch.cuda.is_available(), '| GPU:', torch.cuda.get_device_name(0) if torch.cuda.is_available() else 'None')"

# ==========================================
# 4. CHẠY TRAIN VỚI TMUX (GIỮ PHIÊN CHỐNG RỚT MẠNG)
# ==========================================
# 1. Tạo một session tmux có tên là 'train':
tmux new -s train

# 2. Kích hoạt môi trường:
source .venv/bin/activate

# 3. Khởi chạy huấn luyện:

# --- TUỲ CHỌN 1: MÔ PHỎNG FEDERATED LEARNING (SIMULATION) ---
# Chạy 1 lần:
python -m experiments.run_simulation --config configs/simulation.yaml --run run_1

# Tự động chạy liên tiếp 5 lần (run_1 -> run_5):
for r in run_1 run_2 run_3 run_4 run_5; do
    echo "==================== BAT DAU FL $r ===================="
    python -m experiments.run_simulation --config configs/simulation.yaml --run $r
done

# --- TUỲ CHỌN 2: HUẤN LUYỆN TẬP TRUNG (CENTRALIZED UPPER-BOUND) ---
# Chạy 1 lần:
python -m experiments.train_centralized --dataset pathmnist --model baseline --optimizer sgd --epochs 80 --batch_size 256 --lr 0.01 --run run_1

# Tự động chạy liên tiếp 5 lần Centralized (run_1 -> run_5):
for r in run_1 run_2 run_3 run_4 run_5; do
    echo "==================== BAT DAU CENTRALIZED $r ===================="
    python -m experiments.train_centralized --dataset pathmnist --model baseline --optimizer sgd --epochs 80 --batch_size 256 --lr 0.01 --run $r
done

# 4. Cách thoát ra ngoài mà train vẫn tiếp tục chạy ngầm (Detach):
#    -> Nhấn tổ hợp phím: Ctrl + B
#    -> Sau đó thả tay ra và nhấn phím: D
#    (Lúc này tiến trình train vẫn chạy ngầm trên VM, bạn có thể ngắt SSH hoặc tắt laptop đi ngủ)

# 5. Khi kết nối SSH lại, muốn vào lại xem trực tiếp tiến trình:
tmux attach -t train
